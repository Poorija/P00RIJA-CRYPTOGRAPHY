# Metadata Implementation Plan


**Goal:** Let a person see, edit and remove the metadata in an image, video or audio file — in a tab of its own, and automatically on everything Secure Chat sends.

**Architecture:** One dependency-free module, `js/metadata.js`, that takes bytes and returns bytes. It knows nothing about the DOM or about chat, so it can be tested on its own. Container parsers sit behind one interface, so adding PDF later means adding a file rather than changing callers. Two consumers: a new tab, and one call in the Secure Chat send path.

**Tech Stack:** Plain browser JavaScript, no build step, no external libraries (the page's CSP forbids them and the app must work offline). Playwright for the end-to-end suite, matching every other suite in `tests/e2e/`.

**Spec:** `docs/specs/2026-09-04-metadata-design.md`

## Global Constraints

- **No external dependencies.** The CSP blocks every CDN except for scripts, and the app ships offline. Parsers are written here.
- **No re-encoding.** Pixels and audio samples are copied through byte-for-byte. A stripped file differs from the original only where metadata was.
- **Never claim more than was done.** An unrecognised format returns `supported: false` and the original bytes. It is never described as cleaned.
- **On any failure the original file is returned.** Corrupt input, an unwritable field, a strip that would produce an unreadable file — all return the input unchanged and report the problem.
- **Bilingual, Persian first.** Every user-visible string uses the existing `t(fa, en)` helper in `js/chat.js` or the `translations` object in `js/app.js`, matching the file it lives in.
- **Escaping.** All rendered values pass through `escapeHTML`. Metadata is attacker-controlled text.
- **Existing file conventions.** `js/app.js` and `js/chat.js` are not indented inside top-level blocks; `js/metadata.js` is a new file and uses ordinary two-space indentation, like `js/dialogs.js`.
- **Cache tag.** Any change to `index.html`, `sw.js` or `js/app.js` requires bumping the `?v=` tag in all three, per `HANDOFF.md`.

```javascript
'use strict';
/* What a file says about itself.
 *
 * A photograph from a phone carries where it was taken, on what, and when.
 * Encrypting the photograph and sending those coordinates beside it protects
 * the wrong thing, so this reads them, lets them be edited, and takes them out.
 *
 * No dependencies: the page's CSP blocks external scripts and the app has to
 * work with no network at all. No DOM, no knowledge of chat - bytes in, bytes
 * or a description out - so it can be tested on its own and called from
 * anywhere.
 *
 * One reader per container, all answering the same three questions, so PDF can
 * be added later by adding a file rather than by changing any caller. */
(function (global) {
  /* What a field would tell somebody, rather than how big it is. This is what
     lets the interface say which lines matter without the reader having to know
     what GPSLatitudeRef means. */
  const RISK = { LOCATION: 'location', DEVICE: 'device', IDENTITY: 'identity', TIME: 'time', OTHER: 'other' };

  const EXIF_TAGS = {
    0x010f: { label: 'Make', risk: RISK.DEVICE },
    0x0110: { label: 'Model', risk: RISK.DEVICE },
    0x0131: { label: 'Software', risk: RISK.DEVICE },
    0x0132: { label: 'DateTime', risk: RISK.TIME },
    0x013b: { label: 'Artist', risk: RISK.IDENTITY },
    0x8298: { label: 'Copyright', risk: RISK.IDENTITY },
    0xa430: { label: 'CameraOwnerName', risk: RISK.IDENTITY },
    0xa431: { label: 'BodySerialNumber', risk: RISK.DEVICE },
    0x8825: { label: 'GPSInfo', risk: RISK.LOCATION },
  };
  const GPS_TAGS = {
    0x0000: 'GPSVersionID', 0x0001: 'GPSLatitudeRef', 0x0002: 'GPSLatitude',
    0x0003: 'GPSLongitudeRef', 0x0004: 'GPSLongitude', 0x0005: 'GPSAltitudeRef',
    0x0006: 'GPSAltitude', 0x0007: 'GPSTimeStamp', 0x001d: 'GPSDateStamp',
  };

  const bytesOf = async (input) => new Uint8Array(await input.arrayBuffer());
  const ascii = (view, start, length) => {
    let out = '';
    for (let i = 0; i < length; i += 1) {
      const code = view.getUint8(start + i);
      if (!code) break;
      out += String.fromCharCode(code);
    }
    return out;
  };

  /* --- JPEG ------------------------------------------------------------
     A JPEG is a run of segments, each 0xFF followed by a marker and, for the
     ones that carry anything, a two-byte length. Metadata lives in APP1
     (EXIF and XMP), APP2 (ICC), APP13 (IPTC) and COM; everything else is the
     image. Walking the segments means the pixels are never touched. */
  function jpegSegments(bytes) {
    const segments = [];
    let at = 2;
    while (at + 4 <= bytes.length) {
      if (bytes[at] !== 0xff) break;
      const marker = bytes[at + 1];
      if (marker === 0xd8 || marker === 0xd9) { at += 2; continue; }
      if (marker === 0xda) { segments.push({ marker, start: at, end: bytes.length, scan: true }); break; }
      const length = (bytes[at + 2] << 8) | bytes[at + 3];
      segments.push({ marker, start: at, end: at + 2 + length, bodyAt: at + 4, bodyLength: length - 2 });
      at += 2 + length;
    }
    return segments;
  }
  const JPEG_META_MARKERS = new Set([0xe1, 0xe2, 0xed, 0xee, 0xfe]);

  function readExifBlock(bytes, at, length, fields) {
    if (length < 14) return;
    const view = new DataView(bytes.buffer, bytes.byteOffset + at, length);
    if (ascii(view, 0, 4) !== 'Exif') return;
    const tiff = 6;
    const little = view.getUint16(tiff) === 0x4949;
    const readIfd = (offset, tags, group) => {
      if (offset + 2 > length) return;
      const count = view.getUint16(tiff + offset, little);
      for (let i = 0; i < count; i += 1) {
        const entry = tiff + offset + 2 + (i * 12);
        if (entry + 12 > length) return;
        const tag = view.getUint16(entry, little);
        const type = view.getUint16(entry + 2, little);
        const num = view.getUint32(entry + 4, little);
        const known = tags === EXIF_TAGS ? EXIF_TAGS[tag] : null;
        const label = known ? known.label : (tags === GPS_TAGS ? (GPS_TAGS[tag] || `GPS:${tag}`) : `Tag:${tag}`);
        const risk = group === 'gps' ? RISK.LOCATION : (known ? known.risk : RISK.OTHER);
        let value = '';
        if (type === 2) {
          const inline = num <= 4;
          const at2 = inline ? entry + 8 : tiff + view.getUint32(entry + 8, little);
          if (at2 + num <= length) value = ascii(view, at2, num);
        } else {
          value = String(num <= 4 ? view.getUint32(entry + 8, little) : '…');
        }
        if (tag === 0x8825) {
          readIfd(view.getUint32(entry + 8, little), GPS_TAGS, 'gps');
          continue;
        }
        fields.push({ id: `exif:${group}:${tag}`, group: group === 'gps' ? 'GPS' : 'EXIF', label, value, risk });
      }
    };
    readIfd(view.getUint32(tiff + 4, little), EXIF_TAGS, 'ifd0');
  }

  function readJpeg(bytes) {
    const fields = [];
    jpegSegments(bytes).forEach((segment) => {
      if (segment.scan || !JPEG_META_MARKERS.has(segment.marker)) return;
      if (segment.marker === 0xe1) readExifBlock(bytes, segment.bodyAt, segment.bodyLength, fields);
      else {
        fields.push({
          id: `jpeg:${segment.marker}`,
          group: segment.marker === 0xe2 ? 'ICC' : (segment.marker === 0xed ? 'IPTC' : 'Comment'),
          label: segment.marker === 0xfe ? 'Comment' : 'Block',
          value: `${segment.bodyLength} bytes`,
          risk: RISK.OTHER,
        });
      }
    });
    return { format: 'jpeg', supported: true, fields, warnings: [] };
  }

  async function readMetadata(input) {
    let bytes;
    try {
      bytes = await bytesOf(input);
    } catch (error) {
      return { format: 'unknown', supported: false, fields: [], warnings: ['unreadable'] };
    }
    if (bytes.length > 3 && bytes[0] === 0xff && bytes[1] === 0xd8) {
      try {
        return readJpeg(bytes);
      } catch (error) {
        /* A truncated or malformed file is not a reason to guess. */
        return { format: 'jpeg', supported: false, fields: [], warnings: ['malformed'] };
      }
    }
    return { format: 'unknown', supported: false, fields: [], warnings: ['unrecognised-format'] };
  }

  global.PoorijaMetadata = { readMetadata, RISK };
}(typeof globalThis !== 'undefined' ? globalThis : window));
```

```javascript
  /* Rebuild the file without the segments that carry metadata. Nothing is
     re-encoded: the scan, and every segment that is not metadata, is copied
     across byte for byte, so the picture that comes out is the picture that
     went in. */
  function stripJpeg(bytes) {
    const keep = [];
    const removed = [];
    keep.push(bytes.subarray(0, 2));                    // SOI
    jpegSegments(bytes).forEach((segment) => {
      if (!segment.scan && JPEG_META_MARKERS.has(segment.marker)) {
        removed.push(`jpeg:${segment.marker}`);
        return;
      }
      keep.push(bytes.subarray(segment.start, segment.end));
    });
    const size = keep.reduce((total, part) => total + part.length, 0);
    const out = new Uint8Array(size);
    let at = 0;
    keep.forEach((part) => { out.set(part, at); at += part.length; });
    return { bytes: out, removed };
  }

  async function stripMetadata(input) {
    let bytes;
    try {
      bytes = await bytesOf(input);
    } catch (error) {
      return { blob: input, removed: [], supported: false, format: 'unknown' };
    }
    const untouched = () => ({ blob: input, removed: [], supported: false, format: 'unknown' });
    if (!(bytes.length > 3 && bytes[0] === 0xff && bytes[1] === 0xd8)) return untouched();
    try {
      const result = stripJpeg(bytes);
      /* A strip that produced something that is no longer a JPEG is a bug, and
         the user's file is not the place to find out. */
      if (result.bytes.length < 4 || result.bytes[0] !== 0xff || result.bytes[1] !== 0xd8) return untouched();
      return {
        blob: new Blob([result.bytes], { type: input.type || 'image/jpeg' }),
        removed: result.removed,
        supported: true,
        format: 'jpeg',
      };
    } catch (error) {
      return untouched();
    }
  }
```

```javascript
  /* --- PNG -------------------------------------------------------------
     A PNG is a signature followed by length-prefixed chunks. The text ones and
     eXIf and tIME are metadata; IHDR, IDAT, PLTE and IEND are the picture. */
  const PNG_META = new Set(['tEXt', 'iTXt', 'zTXt', 'eXIf', 'tIME']);
  function pngChunks(bytes) {
    const chunks = [];
    let at = 8;
    const view = new DataView(bytes.buffer, bytes.byteOffset);
    while (at + 12 <= bytes.length) {
      const length = view.getUint32(at);
      const type = ascii(new DataView(bytes.buffer, bytes.byteOffset + at + 4, 4), 0, 4);
      chunks.push({ type, start: at, end: at + 12 + length, bodyAt: at + 8, bodyLength: length });
      at += 12 + length;
      if (type === 'IEND') break;
    }
    return chunks;
  }
  function readPng(bytes) {
    const fields = [];
    pngChunks(bytes).forEach((c) => {
      if (!PNG_META.has(c.type)) return;
      const raw = ascii(new DataView(bytes.buffer, bytes.byteOffset + c.bodyAt, c.bodyLength), 0, Math.min(c.bodyLength, 120));
      fields.push({
        id: `png:${c.type}:${c.start}`, group: 'PNG', label: c.type,
        value: raw.replace(/\0/g, ' '),
        risk: c.type === 'eXIf' ? RISK.LOCATION : (c.type === 'tIME' ? RISK.TIME : RISK.IDENTITY),
      });
    });
    return { format: 'png', supported: true, fields, warnings: [] };
  }
  function stripPng(bytes) {
    const keep = [bytes.subarray(0, 8)];
    const removed = [];
    pngChunks(bytes).forEach((c) => {
      if (PNG_META.has(c.type)) { removed.push(`png:${c.type}:${c.start}`); return; }
      keep.push(bytes.subarray(c.start, c.end));
    });
    return { bytes: concat(keep), removed };
  }

  /* --- WebP ------------------------------------------------------------
     RIFF: a header, then four-character chunks with a length each. */
  const WEBP_META = new Set(['EXIF', 'XMP ', 'ICCP']);
  function riffChunks(bytes) {
    const chunks = [];
    const view = new DataView(bytes.buffer, bytes.byteOffset);
    let at = 12;
    while (at + 8 <= bytes.length) {
      const type = ascii(new DataView(bytes.buffer, bytes.byteOffset + at, 4), 0, 4);
      const length = view.getUint32(at + 4, true);
      const padded = length + (length % 2);
      chunks.push({ type, start: at, end: at + 8 + padded, bodyAt: at + 8, bodyLength: length });
      at += 8 + padded;
    }
    return chunks;
  }
  function readWebp(bytes) {
    const fields = riffChunks(bytes).filter((c) => WEBP_META.has(c.type)).map((c) => ({
      id: `webp:${c.type}:${c.start}`, group: 'WebP', label: c.type.trim(),
      value: `${c.bodyLength} bytes`,
      risk: c.type === 'EXIF' ? RISK.LOCATION : RISK.OTHER,
    }));
    return { format: 'webp', supported: true, fields, warnings: [] };
  }
  function stripWebp(bytes) {
    const keep = [bytes.subarray(0, 12)];
    const removed = [];
    riffChunks(bytes).forEach((c) => {
      if (WEBP_META.has(c.type)) { removed.push(`webp:${c.type}:${c.start}`); return; }
      keep.push(bytes.subarray(c.start, c.end));
    });
    const out = concat(keep);
    /* The RIFF header carries the total size, so it has to be rewritten. */
    new DataView(out.buffer).setUint32(4, out.length - 8, true);
    return { bytes: out, removed };
  }

  /* --- MP4 / MOV / M4A --------------------------------------------------
     Boxes, each a four-byte length and a four-character name. `udta` and
     `meta` are where a phone writes the location and the device; `moov` has to
     be walked because they sit inside it. */
  const MP4_META = new Set(['udta', 'meta']);
  function mp4Boxes(bytes, from, to) {
    const boxes = [];
    const view = new DataView(bytes.buffer, bytes.byteOffset);
    let at = from;
    while (at + 8 <= to) {
      const size = view.getUint32(at);
      const name = ascii(new DataView(bytes.buffer, bytes.byteOffset + at + 4, 4), 0, 4);
      const end = size === 0 ? to : at + size;
      if (size < 8 || end > to) break;
      boxes.push({ name, start: at, end });
      at = end;
    }
    return boxes;
  }
  function readMp4(bytes) {
    const fields = [];
    const walk = (from, to) => {
      mp4Boxes(bytes, from, to).forEach((box) => {
        if (MP4_META.has(box.name)) {
          const raw = ascii(new DataView(bytes.buffer, bytes.byteOffset + box.start + 8, Math.min(box.end - box.start - 8, 160)), 0, 160);
          fields.push({
            id: `mp4:${box.name}:${box.start}`, group: 'MP4', label: box.name,
            value: raw.replace(/[^\x20-\x7e]/g, ' ').trim().slice(0, 120),
            risk: /\+\d|©xyz|loci/.test(raw) ? RISK.LOCATION : RISK.DEVICE,
          });
          return;
        }
        if (box.name === 'moov' || box.name === 'trak') walk(box.start + 8, box.end);
      });
    };
    walk(0, bytes.length);
    return { format: 'mp4', supported: true, fields, warnings: [] };
  }
  function stripMp4(bytes) {
    const removed = [];
    const rebuild = (from, to) => {
      const parts = [];
      mp4Boxes(bytes, from, to).forEach((box) => {
        if (MP4_META.has(box.name)) { removed.push(`mp4:${box.name}:${box.start}`); return; }
        if (box.name === 'moov' || box.name === 'trak') {
          const inner = rebuild(box.start + 8, box.end);
          const head = new Uint8Array(8);
          head.set(bytes.subarray(box.start + 4, box.start + 8), 4);
          new DataView(head.buffer).setUint32(0, inner.length + 8);
          parts.push(head, inner);
          return;
        }
        parts.push(bytes.subarray(box.start, box.end));
      });
      return concat(parts);
    };
    return { bytes: rebuild(0, bytes.length), removed };
  }

  /* --- MP3 --------------------------------------------------------------
     ID3v2 sits at the head with a synchsafe length; ID3v1 is a flat 128 bytes
     at the tail. Everything between them is the audio. */
  function mp3Tags(bytes) {
    const tags = [];
    if (bytes.length > 10 && ascii(new DataView(bytes.buffer, bytes.byteOffset, 3), 0, 3) === 'ID3') {
      const size = ((bytes[6] & 0x7f) << 21) | ((bytes[7] & 0x7f) << 14) | ((bytes[8] & 0x7f) << 7) | (bytes[9] & 0x7f);
      tags.push({ kind: 'id3v2', start: 0, end: 10 + size });
    }
    if (bytes.length > 128 && ascii(new DataView(bytes.buffer, bytes.byteOffset + bytes.length - 128, 3), 0, 3) === 'TAG') {
      tags.push({ kind: 'id3v1', start: bytes.length - 128, end: bytes.length });
    }
    return tags;
  }
  function readMp3(bytes) {
    const fields = mp3Tags(bytes).map((tag) => ({
      id: `mp3:${tag.kind}`, group: 'ID3', label: tag.kind === 'id3v2' ? 'ID3v2' : 'ID3v1',
      value: ascii(new DataView(bytes.buffer, bytes.byteOffset + tag.start, tag.end - tag.start), 0, Math.min(tag.end - tag.start, 160))
        .replace(/[^\x20-\x7e]/g, ' ').trim().slice(0, 120),
      risk: RISK.IDENTITY,
    }));
    return { format: 'mp3', supported: true, fields, warnings: [] };
  }
  function stripMp3(bytes) {
    const tags = mp3Tags(bytes);
    let start = 0;
    let end = bytes.length;
    const removed = [];
    tags.forEach((tag) => {
      removed.push(`mp3:${tag.kind}`);
      if (tag.kind === 'id3v2') start = tag.end;
      else end = tag.start;
    });
    return { bytes: bytes.subarray(start, end), removed };
  }

  function concat(parts) {
    const size = parts.reduce((total, part) => total + part.length, 0);
    const out = new Uint8Array(size);
    let at = 0;
    parts.forEach((part) => { out.set(part, at); at += part.length; });
    return out;
  }

  /* Which container this is, decided from the bytes rather than the name: a
     file called .jpg that is really a PNG must be read as a PNG. */
  function sniff(bytes) {
    if (bytes.length > 3 && bytes[0] === 0xff && bytes[1] === 0xd8) return 'jpeg';
    if (bytes.length > 8 && bytes[0] === 0x89 && ascii(new DataView(bytes.buffer, bytes.byteOffset + 1, 3), 0, 3) === 'PNG') return 'png';
    if (bytes.length > 12
      && ascii(new DataView(bytes.buffer, bytes.byteOffset, 4), 0, 4) === 'RIFF'
      && ascii(new DataView(bytes.buffer, bytes.byteOffset + 8, 4), 0, 4) === 'WEBP') return 'webp';
    if (bytes.length > 12 && ascii(new DataView(bytes.buffer, bytes.byteOffset + 4, 4), 0, 4) === 'ftyp') return 'mp4';
    if (bytes.length > 10 && ascii(new DataView(bytes.buffer, bytes.byteOffset, 3), 0, 3) === 'ID3') return 'mp3';
    if (bytes.length > 2 && bytes[0] === 0xff && (bytes[1] & 0xe0) === 0xe0) return 'mp3';
    return 'unknown';
  }

  const READERS = { jpeg: readJpeg, png: readPng, webp: readWebp, mp4: readMp4, mp3: readMp3 };
  const STRIPPERS = { jpeg: stripJpeg, png: stripPng, webp: stripWebp, mp4: stripMp4, mp3: stripMp3 };
```

```javascript
  async function readMetadata(input) {
    let bytes;
    try { bytes = await bytesOf(input); } catch (error) {
      return { format: 'unknown', supported: false, fields: [], warnings: ['unreadable'] };
    }
    const format = sniff(bytes);
    const reader = READERS[format];
    if (!reader) return { format: 'unknown', supported: false, fields: [], warnings: ['unrecognised-format'] };
    try { return reader(bytes); } catch (error) {
      return { format, supported: false, fields: [], warnings: ['malformed'] };
    }
  }

  async function stripMetadata(input) {
    const untouched = (format) => ({ blob: input, removed: [], supported: false, format: format || 'unknown' });
    let bytes;
    try { bytes = await bytesOf(input); } catch (error) { return untouched(); }
    const format = sniff(bytes);
    const stripper = STRIPPERS[format];
    if (!stripper) return untouched(format);
    try {
      const result = stripper(bytes);
      /* Anything that came out shorter than a header, or no longer starts the
         way its own format must, is a bug - and a user's file is not where a
         bug gets to land. */
      if (!result.bytes || result.bytes.length < 8 || sniff(result.bytes) !== format) return untouched(format);
      return { blob: new Blob([result.bytes], { type: input.type || '' }), removed: result.removed, supported: true, format };
    } catch (error) { return untouched(format); }
  }
```

```javascript
  /* Replacing a value in place. A PNG chunk carries its own length and CRC, so
     one can be rebuilt without disturbing its neighbours; an ID3v2 frame is the
     same shape. EXIF is not: its offsets point at each other, so changing the
     length of one string moves everything after it. For EXIF the offer is to
     clear the field, not to retype it, and `skipped` says which. */
  function writePngField(bytes, id, value) {
    const at = Number(String(id).split(':')[2]);
    const parts = [bytes.subarray(0, 8)];
    let applied = false;
    pngChunks(bytes).forEach((c) => {
      if (c.start !== at) { parts.push(bytes.subarray(c.start, c.end)); return; }
      const data = new TextEncoder().encode(value);
      const name = new TextEncoder().encode(c.type);
      const body = new Uint8Array(name.length + data.length);
      body.set(name, 0); body.set(data, name.length);
      const out = new Uint8Array(8 + body.length + 4);
      new DataView(out.buffer).setUint32(0, data.length);
      out.set(body, 4);
      new DataView(out.buffer).setUint32(4 + body.length, pngCrc(body));
      parts.push(out);
      applied = true;
    });
    return { bytes: concat(parts), applied };
  }
  function pngCrc(buf) {
    let c = ~0;
    for (let i = 0; i < buf.length; i += 1) {
      c ^= buf[i];
      for (let k = 0; k < 8; k += 1) c = (c >>> 1) ^ (0xedb88320 & -(c & 1));
    }
    return ~c >>> 0;
  }

  async function writeMetadata(input, changes) {
    const untouched = (format) => ({ blob: input, applied: [], skipped: Object.keys(changes || {}), supported: false, format: format || 'unknown' });
    let bytes;
    try { bytes = await bytesOf(input); } catch (error) { return untouched(); }
    const format = sniff(bytes);
    if (format !== 'png') return untouched(format);
    const applied = [];
    const skipped = [];
    let working = bytes;
    Object.entries(changes || {}).forEach(([id, value]) => {
      if (!String(id).startsWith('png:')) { skipped.push(id); return; }
      const result = writePngField(working, id, String(value));
      if (!result.applied) { skipped.push(id); return; }
      working = result.bytes;
      applied.push(id);
    });
    if (!applied.length) return untouched(format);
    return { blob: new Blob([working], { type: input.type || 'image/png' }), applied, skipped, supported: true, format };
  }
```

```javascript
      document.querySelectorAll('[data-metadata-input]').forEach((el) => {
        const id = el.getAttribute('data-metadata-input');
        const before = metadataRead.fields.find((f) => f.id === id);
        if (before && String(before.value) !== el.value) changes[id] = el.value;
      });
      if (!Object.keys(changes).length) {
        showNotification(state.language === 'en' ? 'Nothing was changed.' : 'چیزی تغییر نکرده است.', 'info');
        return;

      const out = await window.PoorijaMetadata.writeMetadata(metadataFile, changes);
      if (out.skipped.length) {
        showNotification(state.language === 'en'
          ? `${out.skipped.length} field(s) cannot be rewritten in this format; clear them instead.`
          : `${out.skipped.length} فیلد در این فرمت قابل بازنویسی نیست؛ به‌جایش پاکشان کنید.`, 'warning');
      }
      if (out.applied.length) downloadBlob(out.blob, metadataFile.name.replace(/(\.[^.]+)?$/, '-edited$1'));
    });
  }
  renderMetadataFields();
}
```

```javascript
async function stripFileMetadataForSend(file) {
  if (!window.PoorijaMetadata || !file) return file;
  if (!app().metadataSettings?.().stripOnSend) {
    /* Turned off is a choice, not a reason to stay quiet. Warn once per file
       that actually carries something worth warning about, and name it. */
    try {
      const read = await window.PoorijaMetadata.readMetadata(file);
      const risky = (read.fields || []).filter((f) => f.risk === 'location' || f.risk === 'identity');
      if (risky.length) {
        notify(t(`این فایل ${risky.length} مورد مکان یا هویت دارد و پاک‌سازی خاموش است.`,
          `This file carries ${risky.length} location or identity field(s) and cleaning is off.`), 'warning');
      }
    } catch (error) { /* the warning is a courtesy; its failure must not block a send */ }
    return file;
  }
  try {
    const out = await window.PoorijaMetadata.stripMetadata(file);
    if (!out.supported) return file;
    if (!out.removed.length) return file;
    return new File([out.blob], file.name, { type: file.type, lastModified: Date.now() });
  } catch (error) {
    /* A file that could not be cleaned is sent as it was, not dropped. */
    console.warn('[Chat] metadata could not be stripped; sending the file as it is:', error);
    return file;
  }
}
window.__metadataOnSendProbe = (file) => stripFileMetadataForSend(file);
```
